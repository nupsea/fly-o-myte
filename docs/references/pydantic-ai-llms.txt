# Define a structured output model
from pydantic import BaseModel
from pydantic_ai import Agent

class TravelInsight(BaseModel):
    summary: str          # 3-4 sentence explanation
    price_impact: str     # "higher" | "lower" | "uncertain" | "none"
    event_type: str       # "holiday" | "disruption" | "capacity" | "seasonal"
    confidence: float     # 0.0–1.0
    sources: list[str]    # cited sources

# Create an agent (Claude)
agent = Agent(
    model="claude-haiku-4-5-20251001",
    result_type=TravelInsight,
    system_prompt="You are a travel pricing analyst advising an Australian family...",
)

# Create an agent (Ollama local)
agent = Agent(
    model="ollama:llama3.2",
    result_type=TravelInsight,
    system_prompt="...",
)

# Run the agent (sync)
result = agent.run_sync("Route BNE→SYD, current price $298, average $350...")
insight: TravelInsight = result.data
